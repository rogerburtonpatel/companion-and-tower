%% ------------------------------------------------------- outline scaffolding
%% This document is a typeset OUTLINE, not a draft. Two markers appear in it.
%%
%%   \note{...}  an editorial note from the typesetting pass: something to add,
%%               a claim the data contradicts, a decision to make. Delete each
%%               one as it is resolved. None of it is paper prose.
%%   \gap{...}   a placeholder in the outline itself, where the author left a
%%               blank or wrote "<fill in>".
%%
%% Set \draftfalse in this file to hide every marker and see the page count.
\usepackage[most]{tcolorbox}
\definecolor{noteback}{HTML}{FFF6E5}
\definecolor{noterule}{HTML}{E0A040}
\definecolor{gapback}{HTML}{EAF2FB}
\definecolor{gaprule}{HTML}{4A78B8}

\newcommand{\note}[1]{\ifdraft
  \begin{tcolorbox}[breakable, enhanced, boxrule=0pt, left=5pt, right=5pt,
    top=3pt, bottom=3pt, arc=1pt, colback=noteback, borderline west={2pt}{0pt}{noterule}]
  {\footnotesize\textsf{\textbf{note.}} \footnotesize #1}\end{tcolorbox}\fi}

%% Environment form, for a note that has to contain a listing: a verbatim
%% environment cannot appear inside a macro argument.
\newenvironment{notebox}{\ifdraft
  \begin{tcolorbox}[breakable, enhanced, boxrule=0pt, left=5pt, right=5pt,
    top=3pt, bottom=3pt, arc=1pt, colback=noteback, borderline west={2pt}{0pt}{noterule}]
  \footnotesize\noindent\textsf{\textbf{note.}} }{\end{tcolorbox}\fi}

\newenvironment{gapbox}{\ifdraft
  \begin{tcolorbox}[breakable, enhanced, boxrule=0pt, left=5pt, right=5pt,
    top=3pt, bottom=3pt, arc=1pt, colback=gapback, borderline west={2pt}{0pt}{gaprule}]
  \footnotesize\noindent\textsf{\textbf{gap.}} }{\end{tcolorbox}\fi}
