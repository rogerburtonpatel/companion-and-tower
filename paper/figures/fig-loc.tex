%% Three measures across the three checkpoints, whole development (main+extra).
%% Small multiples rather than one axis: the three measures differ by two orders
%% of magnitude, and a shared scale would hide the tactic series entirely, which
%% is the series the argument turns on.
%% Every panel is zero-based. The definition and proof panels then show bars of
%% nearly equal height, which is the honest reading: those measures barely move.
%% Data: data/chart.csv, generated from paper-data/measure.py output.
\definecolor{cpPaco}{HTML}{2A78D6}
\definecolor{cpOld}{HTML}{EB6834}
\definecolor{cpNew}{HTML}{1BAF7A}
\definecolor{gridink}{HTML}{D8D8D4}
\definecolor{axisink}{HTML}{52514E}

\pgfplotsset{
  locpanel/.style={
    width=0.30\textwidth, height=4.2cm,
    ybar, bar width=16pt, bar shift=0pt,
    xmin=-0.72, xmax=2.72,
    xtick={0,1,2},
    xticklabels={paco,old,new},
    xticklabel style={font=\scriptsize, color=axisink, anchor=north, yshift=-1pt},
    yticklabel style={font=\scriptsize, color=axisink},
    scaled y ticks=false,
    ymajorgrids, grid style={gridink, line width=0.4pt},
    axis line style={axisink, line width=0.4pt},
    tick style={axisink, major tick length=2pt},
    axis x line*=bottom, axis y line*=left,
    title style={font=\small, yshift=-3pt},
    nodes near coords,
    nodes near coords style={font=\tiny, color=axisink,
      /pgf/number format/1000 sep={\,}},
    enlarge y limits={upper, value=0.22},
    ymin=0,
  }
}

\newcommand{\locbar}[3]{%
  \addplot[draw=none, fill=#1] table [col sep=comma, x=idx, y=#3,
      restrict expr to domain={\thisrow{idx}}{#2:#2}] {data/chart.csv};}

\end{tikzpicture}
\caption{The whole development at the three checkpoints, counting the library
proper and its four downstream users together. Each panel carries its own scale,
because the three measures differ by two orders of magnitude, and every panel is
zero-based. Definition lines exclude tactic definitions, which the first panel
counts. Definitions and proofs barely move across the three checkpoints.
Tactic code, the measure enhanced coinduction was expected to reduce, almost
triples at the first port and returns below the \paco{} baseline only after the
library was rewritten.}
