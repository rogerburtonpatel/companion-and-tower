%% A listings language for Rocq, covering the vernacular and the tactic
%% vocabulary this paper actually shows.
\definecolor{kwcolor}{RGB}{20,60,140}
\definecolor{tacticcolor}{RGB}{140,40,20}
\definecolor{commentcolor}{RGB}{90,110,90}

\lstdefinelanguage{Rocq}{
  morekeywords={Definition,Fixpoint,CoFixpoint,Inductive,CoInductive,Variant,
    Lemma,Theorem,Proposition,Corollary,Instance,Program,Class,Record,
    Section,End,Variable,Variables,Context,Hypothesis,Notation,Infix,
    Require,Import,Export,From,Proof,Qed,Defined,Admitted,Abort,Goal,
    forall,exists,fun,match,with,end,return,let,in,if,then,else,
    Ltac,Tactic,Hint,Resolve,Arguments,Global,Local,Set,Unset},
  morekeywords=[2]{coinduction,accumulate,step,unstep,symmetric,tower,
    induction,monauto,icauto,pcofix,gcofix,ginit,gfinal,gbase,ecofix,einit,
    efinal,ebase,upto,apply,intros,destruct,rewrite,reflexivity,eauto,auto,
    cbn,simpl,unfold,fold,inv,inversion,exact,split,left,right,now,easy},
  sensitive=true,
  morecomment=[n]{(*}{*)},
  morestring=[b]",
}

\lstset{
  language=Rocq,
  basicstyle=\ttfamily\footnotesize,
  keywordstyle=\color{kwcolor}\bfseries,
  keywordstyle=[2]\color{tacticcolor},
  commentstyle=\color{commentcolor}\itshape,
  columns=fullflexible,
  keepspaces=true,
  showstringspaces=false,
  breaklines=true,
  xleftmargin=0.6em,
  literate=
    {->}{{$\rightarrow$}}2
    {=>}{{$\Rightarrow$}}2
    {<=}{{$\leq$}}2
    {==>}{{$\Longrightarrow$}}3
    {forall}{{$\forall$}}1
    {exists}{{$\exists$}}1
    {\\/}{{$\vee$}}1
    {/\\}{{$\wedge$}}1,
}
